1 · Raw read quality control

Checks that the eight raw files are intact and describes them before anything is changed. Checksums are compared against the vendor's MD5SUM_Raw.txt; FastQC/MultiQC report per-file quality; DADA2 plots quality along the read, which sets the truncation lengths in step 3.

Script

scripts/01_qc.sh · 40 lines · view on GitHub
#!/usr/bin/env bash
# Step 1 — Raw read QC: checksum verification, FastQC + MultiQC, per-file read stats,
# and DADA2 quality profiles. Inputs: NEW/<marker>/<site>/*.fastq.gz
set -euo pipefail
cd "$(dirname "$0")/.."

OUT=results/01_qc
mkdir -p "$OUT" work/01_fastqc docs/img

# 1a. MD5 verification against the vendor checksum files
: > "$OUT/md5_check.tsv"
for m in 18S COI; do
  while read -r hash file; do
    path=$(ls NEW/"$m"/*/"$file")
    got=$(md5sum < "$path" | cut -d' ' -f1)
    status=$([ "$got" = "$hash" ] && echo OK || echo FAIL)
    printf '%s\t%s\t%s\n' "$m" "$path" "$status" >> "$OUT/md5_check.tsv"
  done < NEW/"$m"/MD5SUM_Raw.txt
done

# 1b. FastQC on every raw file, summarised with MultiQC
find NEW -name '*.fastq.gz' -print0 | xargs -0 fastqc -q -t 8 --outdir work/01_fastqc
multiqc -q -f -o "$OUT/multiqc" -n multiqc_raw.html work/01_fastqc

# 1c. Read count / length / GC / mean quality per file
printf 'marker\tsite\tsample\tread\treads\tmean_len\tgc_pct\tmean_q\n' > "$OUT/read_stats.tsv"
for f in NEW/*/*/*.fastq.gz; do
  marker=$(echo "$f" | cut -d/ -f2); site=$(echo "$f" | cut -d/ -f3)
  base=$(basename "$f" .fastq.gz); sample=${base%_R*}; read=${base##*_}
  zcat "$f" | awk -v m="$marker" -v s="$site" -v sm="$sample" -v r="$read" '
    BEGIN { for (i = 33; i < 127; i++) ord[sprintf("%c", i)] = i - 33 }
    NR % 4 == 2 { n++; len += length($0); gc += gsub(/[GCgc]/, "") }
    NR % 4 == 0 { for (i = 1; i <= length($0); i++) q += ord[substr($0, i, 1)]; qn += length($0) }
    END { printf "%s\t%s\t%s\t%s\t%d\t%.1f\t%.1f\t%.1f\n", m, s, sm, r, n, len/n, 100*gc/len, q/qn }' \
    >> "$OUT/read_stats.tsv"
done

# 1d. Per-position quality profiles (DADA2)

2 · Primer removal (Cutadapt)

Primer sequence is not biological signal and must be removed before denoising. Each read must begin with its primer; pairs where either primer is missing are discarded as off-target. Primers were identified from the reads themselves:

MarkerForwardReverseRegion

COImlCOIintF GGWACWGGWTGAACWGTWTAYCCYCCjgHCO2198 TANACYTCNGGRTGNCCRAARAAYCALeray fragment, Leray et al. 2013

#!/usr/bin/env bash
# Step 2 — Primer removal with Cutadapt.
# Each read must start with its primer (anchored 5'); pairs where either primer is
# missing are discarded. If a read runs through into the opposite primer, that
# (reverse-complemented) primer and everything after it is trimmed too.
#   18S V4 : E572F  CYGCGGTAATTCCAGCTC     / E1009R   AYGGTATCTRWTCRTCTTYG   (Comeau et al. 2011)
#   COI    : mlCOIintF GGWACWGGWTGAACWGTWTAYCCYCC / jgHCO2198 TAIACYTCIGGRTGICCRAARAAYCA (Leray et al. 2013; I -> N)
set -euo pipefail
cd "$(dirname "$0")/.."

OUT=results/02_primer_removal
mkdir -p "$OUT" work/02_trimmed

revcomp() { echo "$1" | rev | tr ACGTRYKMBVDHN TGCAYRMKVBHDN; }

declare -A FWD=( [18S]=CYGCGGTAATTCCAGCTC [COI]=GGWACWGGWTGAACWGTWTAYCCYCC )
declare -A REV=( [18S]=AYGGTATCTRWTCRTCTTYG [COI]=TANACYTCNGGRTGNCCRAARAAYCA )

printf 'marker\tsample\tpairs_in\tpairs_with_both_primers\tpct_kept\n' > "$OUT/cutadapt_summary.tsv"
for m in 18S COI; do
  F=${FWD[$m]}; R=${REV[$m]}
  mkdir -p "work/02_trimmed/$m"
  for r1 in NEW/"$m"/*/*_R1.fastq.gz; do
    s=$(basename "$r1" _R1.fastq.gz); r2=${r1/_R1/_R2}
    cutadapt -j 8 --discard-untrimmed -e 0.15 --minimum-length 50 \
      -g "^${F};required...$(revcomp "$R");optional" -G "^${R};required...$(revcomp "$F");optional" \
      --json "$OUT/${m}_${s}.cutadapt.json" \
      -o "work/02_trimmed/$m/${s}_R1.fastq.gz" -p "work/02_trimmed/$m/${s}_R2.fastq.gz" \
      "$r1" "$r2" > "$OUT/${m}_${s}.cutadapt.log"
    python3 - "$OUT/${m}_${s}.cutadapt.json" "$m" "$s" >> "$OUT/cutadapt_summary.tsv" <<'PY'
import json, sys
j = json.load(open(sys.argv[1]))["read_counts"]
print(f"{sys.argv[2]}\t{sys.argv[3]}\t{j['input']}\t{j['output']}\t{100*j['output']/j['input']:.1f}")
PY
  done
done

# Post-trim length distribution (input to choosing DADA2 truncation lengths)
printf 'marker\tsample\tread\tlength\tcount\n' > "$OUT/trimmed_lengths.tsv"
for f in work/02_trimmed/*/*.fastq.gz; do
  m=$(basename "$(dirname "$f")"); b=$(basename "$f" .fastq.gz)
  zcat "$f" | awk -v m="$m" -v s="${b%_R*}" -v r="${b##*_}" \
    'NR%4==2{c[length($0)]++} END{for(l in c) printf "%s\t%s\t%s\t%d\t%d\n", m, s, r, l, c[l]}' \
    >> "$OUT/trimmed_lengths.tsv"
done

3 · Denoising into ASVs (DADA2)

Reads are quality-filtered, an error model is learned from the data, and true sequence variants (ASVs) are separated from sequencing errors. Both sites are denoised together (pool = TRUE), so the same organism gets the same ASV in both lakes and the sites can be compared directly. Pairs are then merged and chimeras removed.

scripts/03_dada2.R · 73 lines · view on GitHub
# Step 3 — DADA2: filter/trim, learn errors, denoise both sites JOINTLY (pool = TRUE),
# merge pairs, remove chimeras. One run per marker so ASVs are comparable across sites.
# Usage: Rscript scripts/03_dada2.R <marker> <truncF> <truncR> <maxEE_F> <maxEE_R>
suppressPackageStartupMessages({ library(dada2); library(ggplot2); library(data.table) })

args <- commandArgs(trailingOnly = TRUE)
marker <- args[1]
truncLen <- as.integer(args[2:3])
maxEE <- as.numeric(args[4:5])
set.seed(2026)

inDir <- file.path("work/02_trimmed", marker)
filtDir <- file.path("work/03_filtered", marker)
out <- file.path("results/03_dada2", marker)
dir.create(filtDir, recursive = TRUE, showWarnings = FALSE)
dir.create(out, recursive = TRUE, showWarnings = FALSE)

fnF <- sort(list.files(inDir, "_R1.fastq.gz$", full.names = TRUE))
fnR <- sort(list.files(inDir, "_R2.fastq.gz$", full.names = TRUE))
samples <- sub("_R1.fastq.gz$", "", basename(fnF))
filtF <- file.path(filtDir, paste0(samples, "_F.fastq.gz"))
filtR <- file.path(filtDir, paste0(samples, "_R.fastq.gz"))
names(filtF) <- names(filtR) <- samples

# 3a. Quality filtering
filt <- filterAndTrim(fnF, filtF, fnR, filtR, truncLen = truncLen, maxEE = maxEE,
                      maxN = 0, truncQ = 2, rm.phix = TRUE, compress = TRUE, multithread = TRUE)

# 3b. Error models
errF <- learnErrors(filtF, multithread = TRUE, randomize = TRUE, verbose = FALSE)
errR <- learnErrors(filtR, multithread = TRUE, randomize = TRUE, verbose = FALSE)
for (d in list(list(errF, "R1"), list(errR, "R2"))) {
  p <- plotErrors(d[[1]], nominalQ = TRUE) +
    ggtitle(sprintf("%s %s — learned error rates (points) vs fitted model (black line)", marker, d[[2]]))
  ggsave(sprintf("docs/img/03_errors_%s_%s.png", marker, d[[2]]), p, width = 9, height = 7, dpi = 110)
}

# 3c. Joint denoising of both sites
dadaF <- dada(filtF, err = errF, pool = TRUE, multithread = TRUE, verbose = FALSE)
dadaR <- dada(filtR, err = errR, pool = TRUE, multithread = TRUE, verbose = FALSE)

# 3d. Merge pairs, build table, remove chimeras
mergers <- mergePairs(dadaF, filtF, dadaR, filtR, minOverlap = 12, maxMismatch = 0)
seqtab <- makeSequenceTable(mergers)
nochim <- removeBimeraDenovo(seqtab, method = "consensus", multithread = TRUE)

# 3e. Outputs: ASV table, ASV FASTA, length table, read tracking
getN <- function(x) sum(getUniques(x))
track <- data.table(marker = marker, sample = samples,
                    input = filt[, 1], filtered = filt[, 2],
                    denoisedF = sapply(dadaF, getN), denoisedR = sapply(dadaR, getN),
                    merged = sapply(mergers, getN), nonchim = rowSums(nochim))
fwrite(track, file.path(out, "track.tsv"), sep = "\t")

asv_seqs <- colnames(nochim)
asv_ids <- sprintf("%s_ASV%04d", marker, seq_along(asv_seqs))
counts <- data.table(asv = asv_ids, t(nochim))
setnames(counts, c("asv", rownames(nochim)))
counts[, length := nchar(asv_seqs)][, sequence := asv_seqs]
fwrite(counts, file.path(out, "asv_table.tsv"), sep = "\t")
writeLines(paste0(">", asv_ids, "\n", asv_seqs), file.path(out, "asv_seqs.fasta"))

lens <- data.table(length = nchar(colnames(seqtab)), reads = colSums(seqtab),
                   chimera = !colnames(seqtab) %in% asv_seqs)
fwrite(lens[, .(asvs = .N, reads = sum(reads)), by = .(length, chimera)][order(length)],
       file.path(out, "merged_lengths.tsv"), sep = "\t")

summary <- list(marker = marker, truncLen = truncLen, maxEE = maxEE,
                asvs_before_chimera = ncol(seqtab), asvs_final = ncol(nochim),
                chimeric_read_pct = round(100 * (1 - sum(nochim) / sum(seqtab)), 2))
jsonlite::write_json(summary, file.path(out, "summary.json"), auto_unbox = TRUE, pretty = TRUE)
print(track); str(summary)

4 · ASV curation

Removes ASVs that cannot be the targeted gene.

  • COI is protein-coding. A genuine ASV differs from 313 bp only by whole codons and translates with no stop codon under the invertebrate mitochondrial code (NCBI table 5). ASVs failing either test are likely pseudogenes (NUMTs) or artefacts. Detected coi coding frame: +2.
  • 18S ASVs are kept within the expected V4 length window, 350–480 bp.

Script

scripts/04_asv_curation.R · 60 lines · view on GitHub
# Step 4 — ASV curation.
#  COI : a real COI ASV is protein-coding. Keep ASVs whose length differs from the 313 bp
#        Leray fragment by whole codons (±10 codons) AND translate with no stop codon
#        under the invertebrate mitochondrial code (NCBI table 5). Others are likely
#        NUMTs / pseudogenes or artefacts.
#  18S : keep ASVs within the expected V4 (E572F–E1009R) length window, 350–480 bp.
suppressPackageStartupMessages({ library(Biostrings); library(data.table); library(ggplot2) })

out <- "results/04_curation"
dir.create(out, recursive = TRUE, showWarnings = FALSE)
read_asv <- function(m) fread(file.path("results/03_dada2", m, "asv_table.tsv"))

# ---- COI ----
coi <- read_asv("COI")
dna <- DNAStringSet(coi$sequence)
code5 <- getGeneticCode("5")
stops_in_frame <- sapply(1:3, function(f)
  vcountPattern("*", suppressWarnings(translate(subseq(dna, f), genetic.code = code5,
                                                if.fuzzy.codon = "X"))))
# The coding frame is the one that is stop-free for the most reads
frame <- which.max(colSums((stops_in_frame == 0) * (coi$W1 + coi$W2)))
coi[, `:=`(stops = stops_in_frame[, frame],
           codon_ok = abs(length - 313) %% 3 == 0 & abs(length - 313) <= 30)]
coi[, keep := codon_ok & stops == 0]
coi[, reason := fifelse(keep, "kept", fifelse(!codon_ok, "length not codon-compatible", "stop codon in frame"))]

# ---- 18S ----
s18 <- read_asv("18S")
s18[, keep := length >= 350 & length <= 480]
s18[, reason := fifelse(keep, "kept", "outside 350–480 bp")]

# ---- Write curated tables + summary ----
summ <- rbindlist(lapply(list(COI = coi, `18S` = s18), function(d)
  d[, .(asvs = .N, W1_reads = sum(W1), W2_reads = sum(W2)), by = reason]), idcol = "marker")
fwrite(summ, file.path(out, "curation_summary.tsv"), sep = "\t")
fwrite(coi[, .(asv, length, stops, reason)], file.path(out, "COI_frame_check.tsv"), sep = "\t")
cat(sprintf("COI coding frame: +%d\n", frame), file = file.path(out, "COI_frame.txt"))

for (m in c("COI", "18S")) {
  d <- if (m == "COI") coi else s18
  k <- d[keep == TRUE]
  fwrite(k[, .(asv, W1, W2, length, sequence)], file.path(out, sprintf("%s_asv_table_curated.tsv", m)), sep = "\t")
  writeLines(paste0(">", k$asv, "\n", k$sequence), file.path(out, sprintf("%s_asv_seqs_curated.fasta", m)))
}

# ---- Plot: ASV length vs reads, coloured by curation outcome ----
both <- rbind(coi[, .(marker = "COI", length, reads = W1 + W2, reason)],
              s18[, .(marker = "18S", length, reads = W1 + W2, reason)])
p <- ggplot(both, aes(length, reads, colour = reason)) +
  geom_point(alpha = 0.7, size = 1.8) + facet_wrap(~marker, scales = "free_x") +
  scale_y_log10(labels = scales::comma) +
  scale_colour_manual(values = c(kept = "#2b6c8f", `stop codon in frame` = "#d95f02",
                                 `length not codon-compatible` = "#e7a33e",
                                 `outside 350–480 bp` = "#e7a33e")) +
  labs(x = "ASV length (bp)", y = "Reads (log scale)", colour = NULL,
       title = "ASV curation — each point is one ASV") +
  theme_minimal(base_size = 12) + theme(legend.position = "bottom")
ggsave("docs/img/04_curation.png", p, width = 9, height = 4.8, dpi = 120)
print(summ)

5 · Processing summary

Follows every read from raw file to curated ASV table, counts ASVs per site, and checks with rarefaction whether sequencing depth was enough.

# Step 5 — Processing summary: reads surviving each step, ASVs per site and shared
# between sites, and rarefaction curves (was sequencing depth sufficient?).
suppressPackageStartupMessages({ library(data.table); library(ggplot2); library(vegan) })

out <- "results/05_summary"
dir.create(out, recursive = TRUE, showWarnings = FALSE)
raw <- fread("results/01_qc/read_stats.tsv")[read == "R1", .(marker, sample, raw = reads)]
site_of <- c(W1 = "Chandubi (W1)", W2 = "Deepor Beel (W2)")

rows <- list()
for (m in c("18S", "COI")) {
  tr <- fread(file.path("results/03_dada2", m, "track.tsv"))
  cur <- fread(file.path("results/04_curation", sprintf("%s_asv_table_curated.tsv", m)))
  tr[, curated := c(W1 = sum(cur$W1), W2 = sum(cur$W2))[sample]]
  rows[[m]] <- merge(raw[marker == m], tr, by = c("marker", "sample"))
}
track <- rbindlist(rows)[, .(marker, sample, raw, primers_found = input, filtered,
                             merged, nonchim, curated)]
track[, pct_retained := round(100 * curated / raw, 1)]
fwrite(track, file.path(out, "read_tracking.tsv"), sep = "\t")

tl <- melt(track, id.vars = c("marker", "sample"),
           measure.vars = c("raw", "primers_found", "filtered", "merged", "nonchim", "curated"),
           variable.name = "step", value.name = "reads")
tl[, site := site_of[sample]]
p1 <- ggplot(tl, aes(step, reads, group = site, colour = site)) +
  geom_line(linewidth = 1) + geom_point(size = 2) + facet_wrap(~marker) +
  scale_y_continuous(labels = scales::comma, limits = c(0, NA)) +
  scale_colour_manual(values = c("#2b6c8f", "#c2662d")) +
  labs(x = NULL, y = "Read pairs", colour = NULL, title = "Reads surviving each processing step") +

Visualisation

Observations.

  • 53–57 % (18S) and 70–72 % (COI) of raw read pairs reach the final table, against ~40 % in the vendor's COI processing.
  • All four rarefaction curves flatten: sequencing depth was sufficient for both markers at both sites.
  • With joint denoising, 266 of 345 COI ASVs occur in both lakes (vendor table: 2 of 272), confirming that the vendor's near-zero overlap was an artefact of denoising the samples separately.
  • The two markers disagree on how similar the lakes are. COI: Bray–Curtis 0.235, shared ASVs at similar abundance in both lakes. 18S: Bray–Curtis 0.955, only 142 shared ASVs. The COI overlap does not look like tag-jumping (that leaves a trace count on one side, not matched abundances). It may be a genuinely shared zooplankton pool, or the COI libraries may share template DNA; without extraction/PCR blanks this cannot be told apart. Worth raising with the sequencing provider.
  • One sample per lake: these are descriptions, not tested differences.
